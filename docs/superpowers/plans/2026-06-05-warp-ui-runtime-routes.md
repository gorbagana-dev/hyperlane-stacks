# Warp UI Runtime Multi-Route Configuration Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking. **Before editing any file, read its current content** — line numbers here may have drifted.

**Goal:** Make the gorbagana warp UI config-driven at runtime so one image serves any number of deployed routes (gorbagana-only), with route/chain config generated at deploy time and loaded at startup.

**Architecture:** The UI fetches two files from its own server at startup: `warpRoutes.yaml` (a `WarpCoreConfig`, generated by the deployment layer from deployer output, secret-free, committed) and `chains.yaml` (rendered by the container entrypoint from pod env, so the Helius solana RPC secret and local browser RPCs never reach git). The fork gains a small async runtime-config loader; the stacks repo drops the single-route sentinel/`sed` machinery and the build overlay.

**Tech Stack:** Next.js 15 / TypeScript / Vitest (fork); Docker, POSIX sh, Ansible, Python+pytest (stacks).

**Reference spec:** `docs/superpowers/specs/2026-06-05-warp-ui-runtime-routes-design.md`

- [ ] **Step 3: Implement the loader**

```typescript
// src/utils/runtimeConfig.ts
import { ChainMap, ChainMetadata, ChainMetadataSchema, WarpCoreConfig, WarpCoreConfigSchema } from '@hyperlane-xyz/sdk';
import { tryParseJsonOrYaml } from '@hyperlane-xyz/utils';
import { z } from 'zod';
import { logger } from './logger';

// Files written into /public at container start by the deployment layer.
const WARP_ROUTES_URL = '/warpRoutes.yaml';
const CHAINS_URL = '/chains.yaml';

// Fetch a public config file. Returns null if absent (expected in dev / when no
// runtime config is mounted). Throws on network errors other than 404 and on a
// present-but-unparseable body (an unexpected, deploy-time mistake — fail loud).
async function fetchOptionalText(url: string): Promise<string | null> {
  let resp: Response;
  try {
    resp = await fetch(url);
  } catch (e) {
    logger.debug(`Runtime config fetch failed for ${url}, skipping`, e);
    return null;
  }
  if (resp.status === 404) {
    logger.debug(`No runtime config at ${url}, skipping`);
    return null;
  }
  if (!resp.ok) throw new Error(`Runtime config fetch for ${url} failed: ${resp.status}`);
  return resp.text();
}

export async function loadRuntimeWarpRoutes(): Promise<WarpCoreConfig | null> {
  const text = await fetchOptionalText(WARP_ROUTES_URL);
  if (text == null) return null;
  const parsed = tryParseJsonOrYaml(text);
  if (!parsed.success) throw new Error(`Invalid runtime warpRoutes config: ${parsed.error}`);
  const result = WarpCoreConfigSchema.safeParse(parsed.data);
  if (!result.success) throw new Error(`Invalid runtime warpRoutes schema: ${result.error}`);
  return result.data;
}

export async function loadRuntimeChainMetadata(): Promise<ChainMap<ChainMetadata> | null> {
  const text = await fetchOptionalText(CHAINS_URL);
  if (text == null) return null;
  const parsed = tryParseJsonOrYaml(text);
  if (!parsed.success) throw new Error(`Invalid runtime chains config: ${parsed.error}`);
  const result = z.record(ChainMetadataSchema).safeParse(parsed.data);
  if (!result.success) throw new Error(`Invalid runtime chains schema: ${result.error}`);
  return result.data as ChainMap<ChainMetadata>;
}
```

```sh
#!/bin/sh
set -e

# Required env (chain config + secret solana RPC). Routes arrive as a mounted file.
missing=""
for var in GORCHAIN_RPC_URL SOLANA_RPC_URL GORCHAIN_MAILBOX SOLANA_MAILBOX \
           GORCHAIN_DOMAIN_ID SOLANA_DOMAIN_ID GORCHAIN_CHAIN_ID SOLANA_CHAIN_ID; do
  eval val=\$$var
  [ -z "$val" ] && missing="$missing $var"
done
if [ -n "$missing" ]; then
  echo "ERROR: Required environment variables not set:$missing" >&2
  exit 1
fi

WARP_CFG_SRC="/config/warp-ui-config/warpRoutes.yaml"
PUBLIC_DIR="/app/public"

# 1) Route config: copy the deploy-time-generated file the deployment layer mounted.
if [ -f "$WARP_CFG_SRC" ]; then
  cp "$WARP_CFG_SRC" "$PUBLIC_DIR/warpRoutes.yaml"
  echo "Installed warpRoutes.yaml from $WARP_CFG_SRC"
else
  echo "ERROR: $WARP_CFG_SRC not found (warp-ui-config configmap not mounted?)" >&2
  exit 1
fi

# 2) Chain config: render from env so the secret solana RPC never lands in git.
cat > "$PUBLIC_DIR/chains.yaml" <<EOF
gorchain:
  protocol: sealevel
  chainId: ${GORCHAIN_CHAIN_ID}
  domainId: ${GORCHAIN_DOMAIN_ID}
  name: ${GORCHAIN_CHAIN_NAME:-gorchain}
  mailbox: ${GORCHAIN_MAILBOX}
  rpcUrls:
    - http: ${GORCHAIN_RPC_URL}
  nativeToken:
    name: ${GORCHAIN_NATIVE_TOKEN_NAME:-GOR}
    symbol: ${GORCHAIN_NATIVE_TOKEN_SYMBOL:-GOR}
    decimals: ${GORCHAIN_NATIVE_TOKEN_DECIMALS:-9}
  blocks:
    confirmations: 1
    estimateBlockTime: 0.4
    reorgPeriod: 0
solana:
  protocol: sealevel
  chainId: ${SOLANA_CHAIN_ID}
  domainId: ${SOLANA_DOMAIN_ID}
  name: ${SOLANA_CHAIN_NAME:-solana}
  mailbox: ${SOLANA_MAILBOX}
  rpcUrls:
    - http: ${SOLANA_RPC_URL}
  nativeToken:
    name: ${SOLANA_NATIVE_TOKEN_NAME:-SOL}
    symbol: ${SOLANA_NATIVE_TOKEN_SYMBOL:-SOL}
    decimals: ${SOLANA_NATIVE_TOKEN_DECIMALS:-9}
  blocks:
    confirmations: 1
    estimateBlockTime: 0.4
    reorgPeriod: 0
EOF
echo "Rendered chains.yaml"

echo "Starting Next.js standalone server..."
exec node server.js
```

```python
def _build_warp_ui_config(bridge_state_loader, gorchain_mailbox, solana_mailbox):
    """WarpCoreConfig (tokens+options) for every deployed route, for the warp-ui."""
    _STANDARD = {
        "collateral": "SealevelHypCollateral",
        "synthetic": "SealevelHypSynthetic",
        "native": "SealevelHypNative",
    }
    _MAILBOX = {"gorchain": gorchain_mailbox, "solana": solana_mailbox}
    tokens = []
    for route in WARP_ROUTES:
        name = route["name"]
        token_cfg = bridge_state_loader.read_route_token_config(name)["warpRoute"]
        programs = bridge_state_loader.read_route_program_addresses(name)  # {chain: addr}
        sides = {c: s for c, s in token_cfg.items() if c != "name"}
        for chain, side in sides.items():
            other = next(c for c in sides if c != chain)
            entry = {
                "chainName": chain,
                "standard": _STANDARD[side["type"]],
                "name": route.get("display_name", name),
                "symbol": route["stem"].upper(),
                "decimals": side.get("decimals", 9),
                "addressOrDenom": programs[chain],
                "mailbox": _MAILBOX[chain],
                "connections": [{"token": f"sealevel|{other}|{programs[other]}"}],
            }
            if side["type"] == "collateral" and side.get("token"):
                entry["collateralAddressOrDenom"] = side["token"]
            if side["type"] == "synthetic" and side.get("mint"):
                entry["collateralAddressOrDenom"] = side["mint"]
            tokens.append(entry)
    return {"tokens": tokens, "options": {}}
```
